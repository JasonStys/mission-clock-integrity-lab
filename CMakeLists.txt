# C++20 integer clock model by Jason Stys. Native symbols: docs/CODE_INDEX.md.
cmake_minimum_required(VERSION 3.28)
project(MissionClockIntegrityLab VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(CLOCK_SANITIZE "Enable Clang ASan/UBSan" OFF)
option(CLOCK_COVERAGE "Enable GCC coverage" OFF)
if(CLOCK_SANITIZE AND MSVC)
  message(FATAL_ERROR "Sanitizer mode requires Linux Clang")
endif()
if(CLOCK_COVERAGE AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Coverage requires GCC")
endif()
add_library(clock_model native/core.cpp)
target_include_directories(clock_model PUBLIC native)
add_executable(clock_core native/cli.cpp)
target_link_libraries(clock_core PRIVATE clock_model)
add_executable(core_tests native/tests.cpp)
target_link_libraries(core_tests PRIVATE clock_model)
foreach(target clock_model clock_core core_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(CLOCK_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  if(CLOCK_COVERAGE)
    target_compile_options(${target} PRIVATE --coverage -O0 -g)
    target_link_options(${target} PRIVATE --coverage)
  endif()
endforeach()
enable_testing()
add_test(NAME clock_contracts COMMAND core_tests)
